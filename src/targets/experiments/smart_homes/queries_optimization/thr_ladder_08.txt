SELECT * FROM S
WHERE (LOAD as T1; LOAD as T2; LOAD as T3; LOAD as T4; LOAD as T5; LOAD as T6; LOAD as T7; LOAD as T8)
FILTER
    T1[value > 225] AND
    T2[value > 250] AND
    T3[value > 275] AND
    T4[value > 300] AND
    T5[value > 325] AND
    T6[value > 350] AND
    T7[value > 375] AND
    T8[value > 400]
WITHIN 2 [plug_timestamp]
CONSUME BY ANY
LIMIT 1000
