SELECT * FROM S
WHERE (TRIP as Z1)
FILTER
    Z1[fare_amount + tip_amount > total_amount + 100.0] AND
    Z1[fare_amount + tip_amount > total_amount + 100.0] AND
    Z1[fare_amount + tip_amount > total_amount + 100.0] AND
    Z1[fare_amount + tip_amount > total_amount + 100.0]
WITHIN 1000 EVENTS
CONSUME BY ANY
LIMIT 1000
