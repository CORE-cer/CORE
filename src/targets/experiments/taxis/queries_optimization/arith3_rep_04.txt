SELECT * FROM S
WHERE (TRIP as Z1)
FILTER
    Z1[pickup_datetime + long_time_in_secs > dropoff_datetime + 1000] AND
    Z1[pickup_datetime + long_time_in_secs > dropoff_datetime + 1000] AND
    Z1[pickup_datetime + long_time_in_secs > dropoff_datetime + 1000] AND
    Z1[pickup_datetime + long_time_in_secs > dropoff_datetime + 1000]
WITHIN 1000 EVENTS
CONSUME BY ANY
LIMIT 1000
