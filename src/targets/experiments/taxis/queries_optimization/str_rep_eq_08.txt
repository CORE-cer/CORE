SELECT * FROM S
WHERE (TRIP as P1; TRIP as P2; TRIP as P3; TRIP as P4; TRIP as P5; TRIP as P6; TRIP as P7; TRIP as P8; TRIP as G1)
FILTER
    P1[payment_type = 'XXX'] AND
    P2[payment_type = 'XXX'] AND
    P3[payment_type = 'XXX'] AND
    P4[payment_type = 'XXX'] AND
    P5[payment_type = 'XXX'] AND
    P6[payment_type = 'XXX'] AND
    P7[payment_type = 'XXX'] AND
    P8[payment_type = 'XXX'] AND
    G1[total_amount < 0]
WITHIN 1000 EVENTS
CONSUME BY ANY
LIMIT 1000
