FILTER
    Z1[pickup_datetime + long_time_in_secs > dropoff_datetime - 1] AND
    Z2[pickup_datetime + long_time_in_secs > dropoff_datetime - 2] AND
    Z3[pickup_datetime + long_time_in_secs > dropoff_datetime - 3] AND
    Z4[pickup_datetime + long_time_in_secs > dropoff_datetime - 5] AND
    Z5[pickup_datetime + long_time_in_secs > dropoff_datetime - 10] AND
    Z6[pickup_datetime + long_time_in_secs > dropoff_datetime - 20] AND
    Z7[pickup_datetime + long_time_in_secs > dropoff_datetime - 30] AND
    Z8[pickup_datetime + long_time_in_secs > dropoff_datetime - 60] AND
    Z9[pickup_datetime + long_time_in_secs > dropoff_datetime - 120] AND
    Z10[pickup_datetime + long_time_in_secs > dropoff_datetime - 300] AND
    Z11[pickup_datetime + long_time_in_secs > dropoff_datetime - 600] AND
    Z12[pickup_datetime + long_time_in_secs > dropoff_datetime - 1200] AND
    Z13[pickup_datetime + long_time_in_secs > dropoff_datetime - 1800] AND
    Z14[pickup_datetime + long_time_in_secs > dropoff_datetime - 3600] AND
    Z15[pickup_datetime + long_time_in_secs > dropoff_datetime - 7200] AND
    Z16[pickup_datetime + long_time_in_secs > dropoff_datetime - 14400] AND
    G1[total_amount < 0]
WITHIN 1000 EVENTS
