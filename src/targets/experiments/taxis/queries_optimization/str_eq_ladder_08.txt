SELECT * FROM S
WHERE (TRIP as Z1; TRIP as Z2; TRIP as Z3; TRIP as Z4; TRIP as Z5; TRIP as Z6; TRIP as Z7; TRIP as Z8; TRIP as G1)
FILTER
    Z1[pickup_zone = 'East Harlem North'] AND
    Z2[pickup_zone = 'Clinton East'] AND
    Z3[pickup_zone = 'Times Sq/Theatre District'] AND
    Z4[pickup_zone = 'Upper East Side South'] AND
    Z5[pickup_zone = 'Two Bridges/Seward Park'] AND
    Z6[pickup_zone = 'Midtown North'] AND
    Z7[pickup_zone = 'TriBeCa/Civic Center'] AND
    Z8[pickup_zone = 'Yorkville West'] AND
    G1[total_amount < 0]
WITHIN 1000 EVENTS
CONSUME BY ANY
LIMIT 1000
